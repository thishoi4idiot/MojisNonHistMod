on_actions = {
    on_startup = {
        effect = {
            if = {
                limit = {
                    has_game_rule = {
                        rule = free_templates_rule
                        option = ENABLED
                    }
                }
                every_country = {
                    add_ideas = Mojis_Boon
                }
            }
        }
    }

    on_release_as_free = {
        effect = {
            if = {
                limit = {
                    has_game_rule = {
                        rule = free_templates_rule
                        option = ENABLED
                    }
                }
                every_country = {
                    limit  ={
                        NOT = {
                            has_idea = Mojis_Boon
                        }
                    }
                    add_ideas = Mojis_Boon
                }
            }
        }
    }

    on_release_as_puppet = {
        effect = {
            if = {
                limit = {
                    has_game_rule = {
                        rule = free_templates_rule
                        option = ENABLED
                    }
                }
                every_country = {
                    limit  ={
                        NOT = {
                            has_idea = Mojis_Boon
                        }
                    }
                    add_ideas = Mojis_Boon
                }
            }
        }
    }

    on_liberate = {
        effect = {
            if = {
                limit = {
                    has_game_rule = {
                        rule = free_templates_rule
                        option = ENABLED
                    }
                }
                every_country = {
                    limit  ={
                        NOT = {
                            has_idea = Mojis_Boon
                        }
                    }
                    add_ideas = Mojis_Boon
                }
            }
        }
    }
    on_puppet = {
        effect = {
            if = {
                limit = {
                    has_game_rule = {
                        rule = free_templates_rule
                        option = ENABLED
                    }
                }
                every_country = {
                    limit  ={
                        NOT = {
                            has_idea = Mojis_Boon
                        }
                    }
                    add_ideas = Mojis_Boon
                }
            }
        }
    }

    on_government_change = {
        effect = {
            if = {
                limit = {
                    has_game_rule = {
                        rule = free_templates_rule
                        option = ENABLED
                    }
                    NOT = { has_idea = Mojis_Boon }
                }
                every_country = {
                    limit  ={
                        NOT = {
                            has_idea = Mojis_Boon
                        }
                    }
                    add_ideas = Mojis_Boon
                }
            }
        }
    }
}
