add_namespace = SOV_pressure_romania

country_event = {
	id = SOV_pressure_romania.1
	title = SOV_pressure_romania.1.t
	desc = SOV_pressure_romania.1.d
	picture = GFX_report_event_generic_sign_treaty1

	is_triggered_only = yes

	option = {
		name = SOV_pressure_romania.1.a
		ai_chance = {
			base = 50
			modifier = {
				factor = 3
				FROM = { has_country_flag = SOV_bulgaria_in_faction_flag }
			}
		}
		if = {
			limit = {
				FROM = { has_country_flag = SOV_bulgaria_in_faction_flag }
				country_exists = BUL
				owns_state = 77
			}
			77 = {
				transfer_state_to = BUL
			}
		}
		FROM = {
			country_event = { id = SOV_pressure_romania.2 hours = 12 }
		}
	}

	option = {
		name = SOV_pressure_romania.1.b
		ai_chance = {
			base = 50
			modifier = {
				factor = 0.25
				FROM = { has_country_flag = SOV_bulgaria_in_faction_flag }
			}
		}
		if = {
			limit = {
				FROM = { has_country_flag = SOV_bulgaria_in_faction_flag }
				country_exists = BUL
			}
			BUL = {
				create_wargoal = {
					type = take_state_focus
					target = ROOT
					generator = { 77 }
				}
			}
		}
		FROM = {
			create_wargoal = {
				type = take_state_focus
				target = ROOT
				generator = { 78 766 }
			}
			country_event = { id = SOV_pressure_romania.3 hours = 12 }
		}
	}
}
